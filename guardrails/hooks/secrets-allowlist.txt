# A Zod/schema declaration describes the shape of a field, never a value:
# `password: z.string().min(1)` holds no secret.
(password|passwd|pwd|secret|api[_-]?key|access[_-]?token|auth[_-]?token|private[_-]?key)s?:\s*z\.

# An UNQUOTED lowercase identifier or property access is a reference to a value held
# elsewhere (`password: oversized`, `refreshToken: params.refreshToken`), not a literal.
# Quoted values are deliberately NOT covered, so a real inline secret still trips.
(password|passwd|pwd|secret|api[_-]?key|access[_-]?token|auth[_-]?token|private[_-]?key)s?:\s*[a-z][A-Za-z0-9_]*(\.[A-Za-z0-9_]+)*\s*[,}\)]

# A value that is only an env-var NAME is a reference, not the secret itself.
# Covers Helm secretsmanager mappings, where a lowercase secret key maps to an
# uppercase env-var name.
# At least one underscore is required, so an all-caps hex secret still trips.
[:=]\s*[A-Z][A-Z0-9]*(_[A-Z0-9]+)+\s*(#.*)?$
# An empty assignment holds nothing: `password: ''` in a form's defaultValues.
[:=]\s*(''|""|null|undefined)

# SwiftUI empty-string state vars (e.g. @State private var password = "")
var (password|email|token) = ""
# Swift argument label passing the state var (password: password)
password\s*[:=]\s*\$?password\b
# SwiftUI textContentType ternary (.newPassword : .password)
newPassword\s*:\s*\.password
