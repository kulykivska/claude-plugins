# A value that is only an env-var NAME is a reference, not the secret itself.
# Covers Helm secretsmanager mappings, where a lowercase secret key maps to an
# uppercase env-var name.
# At least one underscore is required, so an all-caps hex secret still trips.
[:=]\s*[A-Z][A-Z0-9]*(_[A-Z0-9]+)+\s*(#.*)?$
# An empty assignment holds nothing: `password: ''` in a form's defaultValues.
[:=]\s*(''|""|null|undefined)
